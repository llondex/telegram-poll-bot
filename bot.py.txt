import os
import requests

TOKEN = os.getenv("BOT_TOKEN")
CHAT_ID = os.getenv("CHAT_ID")

url = f"https://api.telegram.org/bot{TOKEN}/sendPoll"

data = {
    "chat_id": CHAT_ID,
    "question": "Привет, будешь на утреннем Прайме? Ориентировочный сбор в 10:00",
    "options": '["Да буду к 10:00","Да буду к 11:00","Да буду к 12:00","Не уверен","нет"]',
    "is_anonymous": False,
    "allows_multiple_answers": False
}

requests.post(url, data=data)
